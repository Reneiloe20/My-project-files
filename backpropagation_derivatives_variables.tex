\section*{Layer 2 --- Output Layer}

\textbf{Step 1: Derivative of loss with respect to prediction}
\[
\frac{\partial \mathcal{L}}{\partial \hat{P}} = \hat{P} - P
\]

\textbf{Step 2: Derivative of Softplus (Sigmoid of $z^2$)}
\[
\frac{\partial \hat{P}}{\partial z^2} = \sigma(z^2) = \frac{e^{z^2}}{1 + e^{z^2}}
\]

\textbf{Step 3: Output delta}
\[
\delta^2 = \frac{\partial \mathcal{L}}{\partial \hat{P}} \cdot \sigma(z^2) = (\hat{P} - P) \cdot \frac{e^{z^2}}{1 + e^{z^2}}
\]

\textbf{Step 4: Gradient with respect to $W^2$}
\[
\frac{\partial \mathcal{L}}{\partial W^2} = \mathbf{a}^1 \cdot \delta^2
\]

\textbf{Step 5: Gradient with respect to $b^2$}
\[
\frac{\partial \mathcal{L}}{\partial b^2} = \delta^2
\]

\section*{Layer 1 --- Hidden Layer}

\textbf{Step 6: Propagate delta backwards through $W^2$}
\[
\frac{\partial \mathcal{L}}{\partial \mathbf{a}^1} = W^2 \cdot \delta^2
\]

\textbf{Step 7: Apply ReLU derivative}
\[
\text{ReLU}'(z) =
\begin{cases}
1 & \text{if } z > 0 \\
0 & \text{if } z \leq 0
\end{cases}
\]
\[
\delta^1 = \frac{\partial \mathcal{L}}{\partial \mathbf{a}^1} \cdot \text{ReLU}'(\mathbf{z}^1)
\]

\textbf{Step 8: Gradient with respect to $W^1$}
\[
\frac{\partial \mathcal{L}}{\partial W^1} = \mathbf{a}^0 \cdot (\delta^1)^T
\]

\textbf{Step 9: Gradient with respect to $\mathbf{b}^1$}
\[
\frac{\partial \mathcal{L}}{\partial \mathbf{b}^1} = \delta^1
\]

\section*{Summary of All Gradients}

\[
\frac{\partial \mathcal{L}}{\partial W^2} = \mathbf{a}^1 \cdot \delta^2, \qquad
\frac{\partial \mathcal{L}}{\partial b^2} = \delta^2
\]

\[
\frac{\partial \mathcal{L}}{\partial W^1} = \mathbf{a}^0 \cdot (\delta^1)^T, \qquad
\frac{\partial \mathcal{L}}{\partial \mathbf{b}^1} = \delta^1
\]

\section*{Weight Update Rule}

For learning rate $\eta$:

\[
W^2 \leftarrow W^2 - \eta \cdot \frac{\partial \mathcal{L}}{\partial W^2}, \qquad
b^2 \leftarrow b^2 - \eta \cdot \frac{\partial \mathcal{L}}{\partial b^2}
\]

\[
W^1 \leftarrow W^1 - \eta \cdot \frac{\partial \mathcal{L}}{\partial W^1}, \qquad
\mathbf{b}^1 \leftarrow \mathbf{b}^1 - \eta \cdot \frac{\partial \mathcal{L}}{\partial \mathbf{b}^1}
\]
